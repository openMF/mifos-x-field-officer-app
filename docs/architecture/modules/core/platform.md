```kotlin
/**
 * Store identity + prompt policy for the in-app review flow.
 *
 * **SoT: `app-profile/`.** The block below is GENERATED by `./gradlew syncForkConfig` from
 * `identity.app_id`, `platforms/apple/apple.yaml#apple.app_store_id`,
 * `platforms/windows/windows.yaml#windows.store_id`, `org.marketing_url`, and `in_app_review.*`.
 * Change those; do not hand-edit the block.
 *
 * ## Why this exists
 * `AppReviewManagerImpl`'s KDoc names the fork step — *"call `AppReview.configure(StoreListing(...))`
 * once at startup with your store ids"* — and nothing did it. Android, iOS and macOS hid the gap
 * behind their native flows, which need no ids; desktop and web reported `canRequestReview == false`
 * forever, so a fork following the docs would render no affordance and see no error explaining why.
 *
 * The ids are not typed here because every one of them already exists in `app-profile` for the
 * DEPLOY side — the Android applicationId, the App Store numeric id the TestFlight lane uploads
 * against, the Partner Center Store ID. A second hand-typed copy is the drift this projection removes.
 *
 * ## Policy vs. the native prompt
 * [enabled] and the three thresholds gate the AUTOMATIC prompt the app shell raises at launch
 * (`AppViewModel`). They are the app's own restraint, layered on top of the OS's: `promptForReview()`
 * reaches a native flow that Play and StoreKit each rate-limit independently, silently showing
 * nothing when they judge it too soon. Neither layer replaces the other — the OS limit is invisible
 * and unconfigurable, so a fork that wants "not before day 3" has to say so here.
 */
object AppReviewConfig {
    // syncForkConfig:app-review:begin — GENERATED from app-profile. Do not hand-edit.
    /** Play Store package — `identity.app_id`. */
    const val PLAY_STORE_PACKAGE: String = "org.mifos.kmp.template"

    /** App Store numeric id — `platforms/apple/apple.yaml#apple.app_store_id`. */
    const val APP_STORE_ID: String = ""

    /** Microsoft Store product id — `platforms/windows/windows.yaml#windows.store_id`. */
    const val MICROSOFT_STORE_PRODUCT_ID: String = ""

    /** Open-web fallback for targets with no store — `org.marketing_url`. */
    const val WEB_URL: String = "https://mifos.org"

    /** `in_app_review.enabled` — false disables the custom prompt entirely. */
    const val ENABLED: Boolean = true

    /** `in_app_review.min_launches`. */
    const val MIN_LAUNCHES: Int = 5

    /** `in_app_review.min_days_since_install`. */
    // … (excerpt)
```
